* ============================================================
* SUBPROGRAM: CRWDUTY
* PURPOSE   : Enforce FAA duty-time regulations
* ============================================================
DEFINE DATA
PARAMETER
1 P-CREW-ID         (A10)
1 P-FLIGHT-KEY      (A20)
1 P-STATUS          (I2)
1 P-STATUS-MSG      (A79)
LOCAL
1 CA VIEW OF CREW-ASSIGNMENT
1 FS VIEW OF FLIGHT-SCHEDULE
1 #BLOCK-HRS        (P5,2)
1 #DUTY-HRS         (P5,2)
1 #NEW-DUTY         (P5,2)
1 #REST-HRS         (P4,2)
1 #LAST-DUTY-END    (A5)
1 #NEXT-DUTY-START  (A5)
END-DEFINE

RESET P-STATUS P-STATUS-MSG

GET CREW-ASSIGNMENT P-CREW-ID

* Get flight block time
FIND FLIGHT-SCHEDULE WITH FS-FLIGHT-KEY = P-FLIGHT-KEY
  COMPUTE #BLOCK-HRS = FS-BLOCK-TIME-MIN / 60
END-FIND

* Check 30-day cumulative
COMPUTE #NEW-DUTY = CA-30DAY-FLIGHT-HRS + #BLOCK-HRS
IF #NEW-DUTY > 100
  MOVE 20 TO P-STATUS
  MOVE '30-DAY LIMIT (100H) WOULD BE EXCEEDED' TO P-STATUS-MSG
  ESCAPE ROUTINE
END-IF

* Check 90-day cumulative
COMPUTE #NEW-DUTY = CA-90DAY-FLIGHT-HRS + #BLOCK-HRS
IF #NEW-DUTY > 270
  MOVE 21 TO P-STATUS
  MOVE '90-DAY LIMIT (270H) WOULD BE EXCEEDED' TO P-STATUS-MSG
  ESCAPE ROUTINE
END-IF

* Check minimum rest before this assignment (10 hours)
COMPUTE #REST-HRS = 10
IF CA-LAST-REST-DATE > *DATX
  MOVE 22 TO P-STATUS
  MOVE 'MINIMUM REST PERIOD NOT SATISFIED' TO P-STATUS-MSG
  ESCAPE ROUTINE
END-IF

MOVE 0 TO P-STATUS
MOVE 'CREW DUTY CHECK PASSED' TO P-STATUS-MSG
END
