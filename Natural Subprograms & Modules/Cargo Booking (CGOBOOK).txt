* ============================================================
* PROGRAM : CGOBOOK
* PURPOSE : Cargo shipment booking
* ============================================================
DEFINE DATA LOCAL
1 CS VIEW OF CARGO-SHIPMENT
1 #AWB-PREFIX       (A3) INIT <'180'>
1 #AWB-SERIAL       (P8)
1 #AWB-CHECK        (P1)
1 #AWB-FULL         (A12)
1 #MAX-WEIGHT       (P9,2)
1 #AVAILABLE-WT     (P9,2)
1 #RATE-PER-KG      (P6,2)
1 #INDEX            (I4)
END-DEFINE

INPUT USING MAP 'CGOCREATE'

* Generate AWB number
CALLNAT 'AWBGEN' #AWB-PREFIX #AWB-SERIAL #AWB-CHECK
COMPRESS #AWB-PREFIX #AWB-SERIAL #AWB-CHECK INTO #AWB-FULL

* Validate weight against aircraft capacity
FIND FLIGHT-SCHEDULE WITH FS-FLIGHT-KEY = CS-FLIGHT-KEY
  COMPUTE #AVAILABLE-WT = FS-CARGO-CAPACITY-KG - FS-CARGO-USED-KG
END-FIND

IF CS-GROSS-WEIGHT-KG > #AVAILABLE-WT
  WRITE 'INSUFFICIENT CAPACITY. AVAILABLE:' #AVAILABLE-WT 'KG'
  ESCAPE ROUTINE
END-IF

* Compute chargeable weight (volumetric vs actual)
COMPUTE CS-CHARGEABLE-WEIGHT = CS-VOLUME-CBM * 167
IF CS-GROSS-WEIGHT-KG > CS-CHARGEABLE-WEIGHT
  MOVE CS-GROSS-WEIGHT-KG TO CS-CHARGEABLE-WEIGHT
END-IF

* Compute charges based on commodity
CALLNAT 'CGORATE' CS-COMMODITY-CODE CS-ORIGIN CS-DESTINATION
  #RATE-PER-KG
COMPUTE CS-FREIGHT-CHARGES = CS-CHARGEABLE-WEIGHT * #RATE-PER-KG

CS-AWB-NUMBER := #AWB-FULL
CS-STATUS := 'BK'

STORE CARGO-SHIPMENT

* Update flight cargo load
FIND FLIGHT-SCHEDULE WITH FS-FLIGHT-KEY = CS-FLIGHT-KEY
  COMPUTE FS-CARGO-USED-KG = FS-CARGO-USED-KG + CS-GROSS-WEIGHT-KG
  UPDATE
END-FIND

WRITE 'AWB:' CS-AWB-NUMBER
      'CHARGES:' CS-FREIGHT-CHARGES
END
