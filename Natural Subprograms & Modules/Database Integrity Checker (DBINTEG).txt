* ============================================================
* PROGRAM : DBINTEG
* PURPOSE : Verify referential integrity across files
* ============================================================
DEFINE DATA LOCAL
1 BK VIEW OF BOOKING
1 PA VIEW OF PASSENGER
1 FS VIEW OF FLIGHT-SCHEDULE
1 #ORPHAN-CNT     (P6)
1 #MISMATCH-CNT   (P6)
1 #TOTAL-CHECKED  (P8)
END-DEFINE

WRITE '=== DATABASE INTEGRITY CHECK ==='

* Check 1: Booking -> Passenger
FIND BOOKING
  ADD 1 TO #TOTAL-CHECKED
  FIND PASSENGER WITH PA-PASSENGER-ID = BK-PASSENGER-ID
    /* found
  END-FIND
  IF *NUMBER = 0
    WRITE 'ORPHAN BOOKING (no passenger):' BK-PNR-LOCATOR
    ADD 1 TO #ORPHAN-CNT
  END-IF
END-FIND

* Check 2: Booking -> Flight schedule
FIND BOOKING WITH BK-BOOKING-STATUS = 'HK'
  FIND FLIGHT-SCHEDULE WITH FS-FLIGHT-KEY = BK-SEG-FLIGHT-KEY(1)
    /* found
  END-FIND
  IF *NUMBER = 0
    WRITE 'BOOKING REFS MISSING FLIGHT:' BK-PNR-LOCATOR
    ADD 1 TO #MISMATCH-CNT
  END-IF
END-FIND

* Check 3: Seat counts match
FIND FLIGHT-SCHEDULE
  FIND BOOKING WITH BK-SEG-FLIGHT-KEY(1) = FS-FLIGHT-KEY
                AND BK-BOOKING-STATUS = 'HK'
  END-FIND
  IF *NUMBER <> FS-SEATS-SOLD
    WRITE 'SEAT COUNT MISMATCH:' FS-FLIGHT-KEY
          'STORED:' FS-SEATS-SOLD 'ACTUAL:' *NUMBER
    ADD 1 TO #MISMATCH-CNT
  END-IF
END-FIND

WRITE '=== INTEGRITY CHECK COMPLETE ==='
WRITE 'CHECKED:' #TOTAL-CHECKED
WRITE 'ORPHANS:' #ORPHAN-CNT
WRITE 'MISMATCHES:' #MISMATCH-CNT
END
