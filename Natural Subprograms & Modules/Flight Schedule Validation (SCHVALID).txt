* ============================================================
* SUBPROGRAM: SCHVALID
* PURPOSE   : Validate flight schedule against business rules
* ============================================================
DEFINE DATA
PARAMETER
1 P-FLIGHT-KEY       (A20)
1 P-RETURN-CODE      (I2)
1 P-RETURN-MSG       (A79)
LOCAL
1 FS VIEW OF FLIGHT-SCHEDULE
1 #CUR-DATE         (A10)
1 #CUR-TIME         (A5)
1 #DEP-DT           (A19)
END-DEFINE

RESET P-RETURN-CODE P-RETURN-MSG
MOVE *DATX TO #CUR-DATE
MOVE *TIMX TO #CUR-TIME

FIND FLIGHT-SCHEDULE WITH FS-FLIGHT-KEY = P-FLIGHT-KEY
  IF FS-FLIGHT-STATUS = 'CN'
    MOVE 8 TO P-RETURN-CODE
    MOVE 'FLIGHT IS CANCELLED' TO P-RETURN-MSG
    ESCAPE ROUTINE
  END-IF

  IF FS-FLIGHT-STATUS = 'DP'
    MOVE 9 TO P-RETURN-CODE
    MOVE 'FLIGHT HAS DEPARTED' TO P-RETURN-MSG
    ESCAPE ROUTINE
  END-IF

  IF FS-SEATS-AVAILABLE <= 0
    MOVE 10 TO P-RETURN-CODE
    MOVE 'NO SEATS AVAILABLE' TO P-RETURN-MSG
    ESCAPE ROUTINE
  END-IF

  IF FS-CREW-ASSIGNED-FLAG = 'N'
    MOVE 11 TO P-RETURN-CODE
    MOVE 'CREW NOT YET ASSIGNED' TO P-RETURN-MSG
  END-IF
END-FIND

IF P-RETURN-CODE = 0
  MOVE 'VALIDATION PASSED' TO P-RETURN-MSG
END-IF
END
