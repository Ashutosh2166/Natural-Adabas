* ============================================================
* SUBPROGRAM: BRENGINE
* PURPOSE   : Rule engine for booking policy enforcement
* ============================================================
DEFINE DATA
PARAMETER
1 P-RULE-ID         (A6)
1 P-CONTEXT         (A100)
1 P-RESULT          (A1)  /* Y/N
1 P-MSG             (A79)
LOCAL
1 #PAX-AGE         (I3)
1 #DAYS-TO-DEPART  (I4)
1 #LOYALTY-TIER    (A2)
1 #FARE-AMOUNT     (P9,2)
1 #CABIN           (A1)
1 #TODAY           (A10)
1 #DEP-DATE        (A10)
1 #IS-REFUNDABLE   (L)
END-DEFINE

RESET P-RESULT P-MSG
CALLNAT 'CTXPARS' P-CONTEXT #PAX-AGE #DAYS-TO-DEPART
  #LOYALTY-TIER #FARE-AMOUNT #CABIN #DEP-DATE
MOVE *DATX TO #TODAY

DECIDE ON FIRST VALUE OF P-RULE-ID
  * RULE: Unaccompanied minor
  VALUE 'RULE01'
    IF #PAX-AGE < 12 AND #PAX-AGE >= 5
      MOVE 'Y' TO P-RESULT
      MOVE 'UNACCOMPANIED MINOR FEE APPLIES ($150)' TO P-MSG
    ELSE IF #PAX-AGE < 5
      MOVE 'N' TO P-RESULT
      MOVE 'CHILD UNDER 5 MUST TRAVEL WITH ADULT' TO P-MSG
    ELSE
      MOVE 'Y' TO P-RESULT
      MOVE 'NO SPECIAL HANDLING' TO P-MSG
    END-IF

  * RULE: Same-day change policy
  VALUE 'RULE02'
    CALLNAT 'DATDIFF' #TODAY #DEP-DATE #DAYS-TO-DEPART
    IF #DAYS-TO-DEPART = 0
      IF #LOYALTY-TIER = 'PL' OR = 'DI'
        MOVE 'Y' TO P-RESULT
        MOVE 'ELITE: FREE SAME-DAY CHANGE' TO P-MSG
      ELSE
        MOVE 'Y' TO P-RESULT
        MOVE 'SAME-DAY CHANGE FEE: $200' TO P-MSG
      END-IF
    END-IF

  * RULE: Refund eligibility
  VALUE 'RULE03'
    IF #CABIN = 'F' OR = 'J'
      MOVE TRUE TO #IS-REFUNDABLE
    ELSE IF #DAYS-TO-DEPART > 7
      MOVE TRUE TO #IS-REFUNDABLE
    ELSE
      MOVE FALSE TO #IS-REFUNDABLE
    END-IF
    IF #IS-REFUNDABLE = TRUE
      MOVE 'Y' TO P-RESULT
      MOVE 'REFUND ELIGIBLE' TO P-MSG
    ELSE
      MOVE 'N' TO P-RESULT
      MOVE 'NON-REFUNDABLE FARE' TO P-MSG
    END-IF

  * RULE: Baggage allowance
  VALUE 'RULE04'
    IF #LOYALTY-TIER = 'DI'
      MOVE 'Y' TO P-RESULT
      MOVE '3 X 32KG BAGS ALLOWED' TO P-MSG
    ELSE IF #LOYALTY-TIER = 'PL'
      MOVE 'Y' TO P-RESULT
      MOVE '2 X 32KG BAGS ALLOWED' TO P-MSG
    ELSE IF #LOYALTY-TIER = 'GL'
      MOVE 'Y' TO P-RESULT
      MOVE '2 X 23KG BAGS ALLOWED' TO P-MSG
    ELSE IF #CABIN = 'F' OR = 'J'
      MOVE 'Y' TO P-RESULT
      MOVE '2 X 32KG BAGS ALLOWED' TO P-MSG
    ELSE
      MOVE 'Y' TO P-RESULT
      MOVE '1 X 23KG BAG ALLOWED' TO P-MSG
    END-IF

  * RULE: Advance purchase requirement
  VALUE 'RULE05'
    CALLNAT 'DATDIFF' #TODAY #DEP-DATE #DAYS-TO-DEPART
    IF #FARE-AMOUNT < 300 AND #DAYS-TO-DEPART < 14
      MOVE 'N' TO P-RESULT
      MOVE 'LOW FARE REQUIRES 14-DAY ADVANCE PURCHASE' TO P-MSG
    ELSE
      MOVE 'Y' TO P-RESULT
      MOVE 'ADVANCE PURCHASE OK' TO P-MSG
    END-IF

  * RULE: Upgrade eligibility
  VALUE 'RULE06'
    IF #CABIN = 'Y' AND #LOYALTY-TIER = 'DI'
      MOVE 'Y' TO P-RESULT
      MOVE 'ELITE UPGRADE AVAILABLE' TO P-MSG
    ELSE IF #CABIN = 'Y' AND #LOYALTY-TIER = 'PL'
      AND #FARE-AMOUNT >= 500
      MOVE 'Y' TO P-RESULT
      MOVE 'UPGRADE AVAILABLE WITH COPAY' TO P-MSG
    ELSE
      MOVE 'N' TO P-RESULT
      MOVE 'NO UPGRADE AVAILABLE' TO P-MSG
    END-IF

  * RULE: Corporate travel compliance
  VALUE 'RULE07'
    IF #CABIN = 'F'
      MOVE 'N' TO P-RESULT
      MOVE 'CORP POLICY: FIRST CLASS NOT ALLOWED' TO P-MSG
    ELSE IF #CABIN = 'J' AND #FARE-AMOUNT > 3000
      MOVE 'N' TO P-RESULT
      MOVE 'CORP POLICY: BUSINESS FARE EXCEEDS LIMIT' TO P-MSG
    ELSE
      MOVE 'Y' TO P-RESULT
      MOVE 'CORP POLICY COMPLIANT' TO P-MSG
    END-IF

  * RULE: International document check
  VALUE 'RULE08'
    MOVE 'Y' TO P-RESULT
    MOVE 'VERIFY PASSPORT VALIDITY >6 MONTHS PAST RETURN' TO P-MSG

  * RULE: Pet travel
  VALUE 'RULE09'
    MOVE 'Y' TO P-RESULT
    MOVE 'PET IN CABIN: MAX 8KG; CARGO: >8KG' TO P-MSG

  * RULE: Sporting equipment
  VALUE 'RULE10'
    MOVE 'Y' TO P-RESULT
    MOVE 'SPORTING EQUIP FEE: $75-150 PER ITEM' TO P-MSG

  VALUE 'RULE11'
    MOVE 'Y' TO P-RESULT
    MOVE 'MEDICAL CLEARANCE REQUIRED IF >28 WEEKS' TO P-MSG

  VALUE 'RULE12'
    MOVE 'Y' TO P-RESULT
    MOVE 'EMERGENCY EXIT SEAT: MIN AGE 15, ABLE-BODIED' TO P-MSG

  VALUE 'RULE13'
    MOVE 'Y' TO P-RESULT
    MOVE 'BULKHEAD SEAT: NO INFANTS PER FAA' TO P-MSG

  * Default
  VALUE 'RULE99'
    MOVE 'N' TO P-RESULT
    MOVE 'UNKNOWN RULE ID' TO P-MSG

  NONE
    MOVE 'N' TO P-RESULT
    MOVE 'RULE NOT FOUND' TO P-MSG
END-DECIDE
END
