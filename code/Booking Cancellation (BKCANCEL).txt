* ============================================================
* PROGRAM : BKCANCEL
* PURPOSE : Cancel booking with refund calculation
* ============================================================
DEFINE DATA LOCAL
1 BK VIEW OF BOOKING
1 #PNR              (A8)
1 #REFUND-AMOUNT    (P12,2)
1 #PENALTY-AMOUNT   (P12,2)
1 #CONFIRM          (A1)
1 #DAYS-TO-DEPART   (I4)
1 #TODAY            (A10)
1 #DEP-DATE         (A10)
END-DEFINE

INPUT 'ENTER PNR TO CANCEL:' #PNR

FIND BOOKING WITH BK-PNR-LOCATOR = #PNR
  DISPLAY 'PNR:' BK-PNR-LOCATOR
          'PASSENGER:' BK-PASSENGER-ID
          'AMOUNT:' BK-TOTAL-AMOUNT BK-CURRENCY
          'STATUS:' BK-BOOKING-STATUS
  INPUT 'CONFIRM CANCELLATION (Y/N):' #CONFIRM
  IF #CONFIRM <> 'Y' AND <> 'y'
    ESCAPE ROUTINE
  END-IF

  * Calculate refund based on fare rules
  PERFORM V-CALC-REFUND

  IF BK-BOOKING-STATUS = 'TK'
    UPDATE BOOKING
      BK-BOOKING-STATUS := 'NO'
      BK-MODIFIED-TS := *TIMX
  ELSE
    UPDATE BOOKING
      BK-BOOKING-STATUS := 'UC'
      BK-MODIFIED-TS := *TIMX
  END-IF

  WRITE 'CANCELLED:' #PNR
        'REFUND:' #REFUND-AMOUNT
        'PENALTY:' #PENALTY-AMOUNT
  CALLNAT 'AUDITLOG' 'BK' 'CANCEL' #PNR
END-FIND

DEFINE SUBROUTINE V-CALC-REFUND
  MOVE BK-BOOKING-DATE TO #TODAY
  * Extract departure date from first segment
  MOVE SUBSTRING(BK-SEG-FLIGHT-KEY(1), 14, 10) TO #DEP-DATE
  CALLNAT 'DATDIFF' #TODAY #DEP-DATE #DAYS-TO-DEPART

  IF #DAYS-TO-DEPART >= 30
    MOVE 0 TO #PENALTY-AMOUNT
  ELSE IF #DAYS-TO-DEPART >= 7
    COMPUTE #PENALTY-AMOUNT = BK-TOTAL-AMOUNT * 0.10
  ELSE
    COMPUTE #PENALTY-AMOUNT = BK-TOTAL-AMOUNT * 0.25
  END-IF

  COMPUTE #REFUND-AMOUNT = BK-TOTAL-AMOUNT - #PENALTY-AMOUNT
END-SUBROUTINE
END
