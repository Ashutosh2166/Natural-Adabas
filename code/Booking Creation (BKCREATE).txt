* ============================================================
* PROGRAM : BKCREATE
* PURPOSE : Create a new PNR / Booking
* ============================================================
DEFINE DATA LOCAL
1 BK VIEW OF BOOKING
END-DEFINE

1 #INPUT-SEGMENTS (1:8)
  2 #SEG-FLIGHT-KEY   (A20)
  2 #SEG-CLASS        (A1)
  2 #SEG-DATE         (A10)
1 #SEG-COUNT         (I2)
1 #VALID-FLAG        (L)
1 #FARE-RESULT       (P9,2)
1 #TOTAL-FARE        (P12,2)
1 #ERROR-MSG         (A79)
1 #RETURN-CODE       (I2)

* --- Get passenger details first ---
CALLNAT 'PAGETDTL' BK-PASSENGER-ID
  #RETURN-CODE
IF #RETURN-CODE <> 0
  MOVE 'PASSENGER LOOKUP FAILED' TO #ERROR-MSG
  ESCAPE ROUTINE
END-IF

* --- Prompt user to enter itinerary ---
INPUT USING MAP 'BKSEGIN'
  #SEG-COUNT
  #INPUT-SEGMENTS(*)

* --- Validate segments ---
CALLNAT 'SEGVALID' #INPUT-SEGMENTS(*) #SEG-COUNT
  #VALID-FLAG #ERROR-MSG
IF #VALID-FLAG = FALSE
  REINPUT 'SEGMENT VALIDATION FAILED: ' #ERROR-MSG
END-IF

* --- Calculate fare for each segment ---
PERFORM V-COMPUTE-FARES

* --- Generate PNR locator ---
CALLNAT 'PNRGEN' BK-PNR-LOCATOR
BK-BOOKING-DATE := *DATX
BK-BOOKING-STATUS := 'HK'
BK-TOTAL-AMOUNT := #TOTAL-FARE
BK-CURRENCY := 'USD'
BK-CREATED-TS := *TIMX

* --- Insert into ADABAS ---
STORE BOOKING
  USING #SEG-COUNT
IF *ISN = 0
  MOVE 'DATABASE STORE FAILED' TO #ERROR-MSG
  ESCAPE ROUTINE
END-IF

CALLNAT 'AUDITLOG' 'BK' 'INSERT' BK-PNR-LOCATOR
MOVE 'BOOKING CREATED SUCCESSFULLY: ' TO #ERROR-MSG
WRITE #ERROR-MSG BK-PNR-LOCATOR

* ---------- Subroutines ----------
DEFINE SUBROUTINE V-COMPUTE-FARES
  RESET #TOTAL-FARE
  FOR #I = 1 TO #SEG-COUNT
    CALLNAT 'FARECALC'
      #INPUT-SEGMENTS(#I)
      BK-PASSENGER-ID
      #FARE-RESULT
    ADD #FARE-RESULT TO #TOTAL-FARE
  END-FOR
END-SUBROUTINE

END
